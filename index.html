/* ========================= APPLE GAME FULL ========================= */ const APPLE_COLS = 17; const APPLE_ROWS = 10; const APPLE_TOTAL = APPLE_COLS * APPLE_ROWS; const APPLE_LIMIT = 100; let appleBoard = []; let appleSelectedIndexes = []; let appleDragging = false; let applePointerStarted = false; let appleScore = 0; let appleTimeLeft = APPLE_LIMIT; let appleTimer = null; let appleRunning = false; window.showAppleLobby = showAppleLobby; window.startAppleGame = startAppleGame; function showAppleLobby() { showPage("appleGamePage"); document.getElementById("appleLobby").classList.remove("hidden"); document.getElementById("appleGameWrap").classList.add("hidden"); document.getElementById("appleGameOverlay").classList.add("hidden"); stopAppleTimer(); } function startAppleGame() { document.getElementById("appleLobby").classList.add("hidden"); document.getElementById("appleGameWrap").classList.remove("hidden"); document.getElementById("appleGameOverlay").classList.add("hidden"); appleBoard = Array.from({ length: APPLE_TOTAL }, () => ({ value: Math.floor(Math.random() * 9) + 1, removed: false })); appleSelectedIndexes = []; appleScore = 0; appleTimeLeft = APPLE_LIMIT; appleRunning = true; renderAppleBoard(); startAppleTimer(); } function renderAppleBoard() { const boardEl = document.getElementById("appleBoard"); if (!boardEl) return; boardEl.innerHTML = ""; appleBoard.forEach((cell, index) => { const button = document.createElement("button"); button.className = "apple-cell"; button.type = "button"; if (cell.removed) { button.classList.add("removed"); button.disabled = true; button.textContent = ""; } else { button.textContent = cell.value; if (appleSelectedIndexes.includes(index)) { button.classList.add("selected"); } button.addEventListener("pointerdown", (e) => { e.preventDefault(); handleApplePointerDown(index); }); button.addEventListener("pointerenter", () => { handleApplePointerEnter(index); }); } boardEl.appendChild(button); }); boardEl.onpointerup = () => handleApplePointerUp(); updateAppleHud(); } function handleApplePointerDown(index) { if (!appleRunning) return; const cell = appleBoard[index]; if (!cell || cell.removed) return; clearAppleSelection(false); appleDragging = true; applePointerStarted = true; trySelectAppleCell(index); } function handleApplePointerEnter(index) { if (!appleDragging) return; trySelectAppleCell(index); } function handleApplePointerUp() { appleDragging = false; applePointerStarted = false; } function trySelectAppleCell(index) { const cell = appleBoard[index]; if (!cell || cell.removed) return; if (appleSelectedIndexes.includes(index)) return; if (appleSelectedIndexes.length > 0) { const last = appleSelectedIndexes[appleSelectedIndexes.length - 1]; if (!isOrthogonallyAdjacent(last, index)) return; } appleSelectedIndexes.push(index); const total = appleSelectedIndexes.reduce( (sum, i) => sum + appleBoard[i].value, 0 ); if (total > 10) { clearAppleSelection(); return; } if (total === 10) { appleSelectedIndexes.forEach((i) => { appleBoard[i].removed = true; appleScore++; }); clearAppleSelection(false); renderAppleBoard(); return; } renderAppleBoard(); } function clearAppleSelection(render = true) { appleSelectedIndexes = []; if (render) renderAppleBoard(); } function isOrthogonallyAdjacent(a, b) { const ar = Math.floor(a / APPLE_COLS); const ac = a % APPLE_COLS; const br = Math.floor(b / APPLE_COLS); const bc = b % APPLE_COLS; return ( (Math.abs(ar - br) === 1 && ac === bc) || (Math.abs(ac - bc) === 1 && ar === br) ); } function updateAppleHud() { const scoreEl = document.getElementById("appleMiniScoreValue"); const timeEl = document.getElementById("appleTimeText"); const barEl = document.getElementById("appleTimeBar"); if (scoreEl) scoreEl.textContent = appleScore; if (timeEl) timeEl.textContent = appleTimeLeft; if (barEl) { const percent = (appleTimeLeft / APPLE_LIMIT) * 100; barEl.style.height = `${percent}%`; } } function startAppleTimer() { stopAppleTimer(); appleTimer = setInterval(() => { appleTimeLeft--; if (appleTimeLeft <= 0) { appleTimeLeft = 0; finishAppleGame(); } updateAppleHud(); }, 1000); } function stopAppleTimer() { if (appleTimer) { clearInterval(appleTimer); appleTimer = null; } } function finishAppleGame() { stopAppleTimer(); appleRunning = false; document.getElementById("appleFinalScore").textContent = appleScore; document.getElementById("appleGameOverlay").classList.remove("hidden"); } function resetAppleGame() { startAppleGame(); } document.addEventListener("pointerup", () => { appleDragging = false; });
